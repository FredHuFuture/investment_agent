Wilson 95% CI formula (z = 1.96):
```
center = (p_hat + z² / 2n) / (1 + z² / n)
half_width = z * sqrt(p_hat (1 - p_hat) / n + z² / 4n²) / (1 + z² / n)
ci_low  = max(0, center - half_width)
ci_high = min(1, center + half_width)
```

Murphy exact-bin decomposition (Wikipedia / Siegert 2017):
```
N    = sum(n_k)        # total samples
o_bar = sum(n_k * o_k) / N        # base rate
REL  = (1/N) * sum(n_k * (f_k - o_k)²)  # calibration
RES  = (1/N) * sum(n_k * (o_k - o_bar)²) # resolution
UNC  = o_bar * (1 - o_bar)               # uncertainty
Brier = REL - RES + UNC                  # invariant
```

    ```python
    async def compute_reliability_bins(
        self,
        agent_name: str,
        horizon: str = "5d",
        min_per_bin: int = 10,
        max_bins: int = 10,
        fundamentals_provider: str | None = "yfinance",
    ) -> dict[str, Any]:
        """Reliability bin computation — predicted prob × observed frequency with
        Wilson 95% CIs. Reads from backtest_signal_history (consistent with Brier/IC).
        Excludes HOLD signals (one-vs-rest binary, mirrors compute_brier_score).

        Returns:
            {
                "bins": [{bin_lo, bin_hi, n, predicted, observed, ci_low, ci_high, ece_contrib}, ...],
                "n_samples": int,
                "n_bins_used": int,
                "preliminary_calibration": bool,
                "ece": float | None,
            }

        Pitfall 2 mitigation: adaptive bin count + preliminary_calibration flag.
        Pitfall 4 mitigation: filters corpus by fundamentals_provider.
        """
        from sklearn.calibration import calibration_curve  # noqa: lazy import
        import numpy as np

        rows = await self._store.get_backtest_signals_by_agent(
            agent_name, horizon, fundamentals_provider=fundamentals_provider,
        )
        y_true_list: list[int] = []
        y_prob_list: list[float] = []
        for r in rows:
            if r["signal"] == "HOLD":
                continue
            fwd = r.get(f"forward_return_{horizon}") or r.get("forward_return_5d")
            if fwd is None:
                continue
            y_prob = float(r["confidence"]) / 100.0
            if r["signal"] == "BUY":
                y_true_list.append(1 if fwd > 0 else 0)
            else:  # SELL
                y_true_list.append(1 if fwd < 0 else 0)
            y_prob_list.append(y_prob)

        n_samples = len(y_true_list)
        n_bins = _adaptive_bin_count(n_samples, min_per_bin, max_bins)

        if n_samples < min_per_bin * 2:
            return {
                "bins": [],
                "n_samples": n_samples,
                "n_bins_used": 0,
                "preliminary_calibration": True,
                "ece": None,
            }

        y_true_arr = np.asarray(y_true_list, dtype=int)
        y_prob_arr = np.asarray(y_prob_list, dtype=float)

        # Compute bin edges via quantiles to ensure equal-N bins (sklearn strategy='quantile')
        edges = np.quantile(y_prob_arr, np.linspace(0, 1, n_bins + 1))
        # Ensure strictly increasing edges (handle duplicate quantiles)
        edges = np.unique(edges)
        if len(edges) - 1 < 2:
            return {
                "bins": [], "n_samples": n_samples, "n_bins_used": 0,
                "preliminary_calibration": True, "ece": None,
            }

        bin_indices = np.digitize(y_prob_arr, edges[1:-1])  # 0..len(edges)-2
        z = 1.96
        bins_out: list[dict[str, Any]] = []
        for k in range(len(edges) - 1):
            mask = bin_indices == k
            n_k = int(mask.sum())
            if n_k == 0:
                continue
            observed = float(y_true_arr[mask].mean())
            predicted = float(y_prob_arr[mask].mean())
            # Wilson 95% CI
            p_hat = observed
            denom = 1.0 + (z ** 2) / n_k
            center = (p_hat + (z ** 2) / (2 * n_k)) / denom
            half = z * np.sqrt(p_hat * (1 - p_hat) / n_k + (z ** 2) / (4 * n_k ** 2)) / denom
            ci_low = max(0.0, center - half)
            ci_high = min(1.0, center + half)
            ece_contrib = (n_k / n_samples) * abs(predicted - observed)
            bins_out.append({
                "bin_lo": float(edges[k]),
                "bin_hi": float(edges[k + 1]),
                "n": n_k,
                "predicted": predicted,
                "observed": observed,
                "ci_low": float(ci_low),
                "ci_high": float(ci_high),
                "ece_contrib": float(ece_contrib),
            })

        ece = sum(b["ece_contrib"] for b in bins_out)
        min_bin_n = min((b["n"] for b in bins_out), default=0)
        preliminary = len(bins_out) < 5 or min_bin_n < min_per_bin
        return {
            "bins": bins_out,
            "n_samples": n_samples,
            "n_bins_used": len(bins_out),
            "preliminary_calibration": preliminary,
            "ece": float(ece),
        }
    ```

    ```python
    @staticmethod
    def compute_murphy_decomposition(bins_response: dict) -> dict:
        """Murphy 3-component Brier decomposition: REL - RES + UNC ≈ Brier.

        Exact-bin formulas from Murphy 1973 / Wikipedia 'Brier score'.
        Bias-corrected exact-bin is stable at N ≥ 60 per bin (Ferro & Fricker 2012).
        Below threshold, the bins_response.preliminary_calibration flag surfaces.

        Args:
            bins_response: output of compute_reliability_bins, with key 'bins' containing
                           list of {n, predicted, observed, ...} dicts.

        Returns:
            {rel, res, unc, verified_sum} all floats, or all None when bins empty.
        """
        import numpy as np
        bins = bins_response.get("bins", [])
        if not bins:
            return {"rel": None, "res": None, "unc": None, "verified_sum": None}
        n_arr = np.asarray([b["n"] for b in bins], dtype=float)
        f_arr = np.asarray([b["predicted"] for b in bins], dtype=float)
        o_arr = np.asarray([b["observed"] for b in bins], dtype=float)
        N = float(n_arr.sum())
        if N == 0:
            return {"rel": None, "res": None, "unc": None, "verified_sum": None}
        o_bar = float((n_arr * o_arr).sum() / N)
        rel = float((n_arr * (f_arr - o_arr) ** 2).sum() / N)
        res = float((n_arr * (o_arr - o_bar) ** 2).sum() / N)
        unc = float(o_bar * (1.0 - o_bar))
        return {
            "rel": rel,
            "res": res,
            "unc": unc,
            "verified_sum": rel - res + unc,  # ≈ Brier (sanity invariant)
        }
    ```
