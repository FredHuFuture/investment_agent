    ```ts
    export async function getCalibrationAnalytics(
      options?: {
        include_reliability?: boolean;
        fundamentals_provider?: "yfinance" | "simfin";
      }
    ): Promise<CalibrationResponse> {
      const params = new URLSearchParams();
      if (options?.include_reliability) params.set("include_reliability", "true");
      if (options?.fundamentals_provider) {
        params.set("fundamentals_provider", options.fundamentals_provider);
      }
      const qs = params.toString() ? `?${params.toString()}` : "";
      return apiGet<CalibrationResponse>(`/api/v1/analytics/calibration${qs}`);
    }
    ```

    ```tsx
    import {
      ScatterChart, Scatter, XAxis, YAxis, ZAxis, Tooltip, ResponsiveContainer,
      ReferenceLine, ErrorBar,
    } from "recharts";
    import type { ReliabilityBin } from "../../api/types";

    interface ReliabilityPlotData {
      bins: ReliabilityBin[];
      n_samples: number;
      n_bins_used: number;
      preliminary_calibration: boolean;
      ece: number | null;
    }

    interface Props {
      data: ReliabilityPlotData;
      agentName: string;
    }

    export default function ReliabilityPlot({ data, agentName }: Props) {
      if (!data.bins || data.bins.length === 0) {
        return (
          <div
            className="text-xs text-gray-500 italic p-4"
            data-testid={`reliability-plot-empty-${agentName}`}
          >
            No reliability data — corpus too thin (N={data.n_samples}). Run a corpus rebuild on /calibration.
          </div>
        );
      }

      const points = data.bins.map((b) => ({
        x: b.predicted,
        y: b.observed,
        n: b.n,
        errLow: b.observed - b.ci_low,
        errHigh: b.ci_high - b.observed,
      }));

      return (
        <div className="space-y-2" data-testid={`reliability-plot-${agentName}`}>
          {data.preliminary_calibration && (
            <div
              className="text-xs text-amber-400 bg-amber-950/30 rounded px-2 py-1"
              data-testid={`reliability-preliminary-banner-${agentName}`}
            >
              Preliminary calibration — {data.n_samples} samples, {data.n_bins_used} bin
              {data.n_bins_used !== 1 ? "s" : ""}. Need ≥ 200 samples for stable estimates.
            </div>
          )}
          <ResponsiveContainer width="100%" height={280}>
            <ScatterChart margin={{ top: 8, right: 16, bottom: 32, left: 32 }}>
              <XAxis
                type="number"
                dataKey="x"
                domain={[0, 1]}
                name="Predicted prob"
                label={{ value: "Predicted probability", position: "insideBottom", offset: -8 }}
              />
              <YAxis
                type="number"
                dataKey="y"
                domain={[0, 1]}
                name="Observed freq"
                label={{ value: "Observed frequency", angle: -90, position: "insideLeft" }}
              />
              <ZAxis type="number" dataKey="n" range={[40, 400]} name="N samples" />
              <ReferenceLine
                segment={[{ x: 0, y: 0 }, { x: 1, y: 1 }]}
                stroke="#6b7280"
                strokeDasharray="4 4"
              />
              <Scatter name={agentName} data={points} fill="#10b981">
                <ErrorBar dataKey="errLow" direction="y" stroke="#10b981" />
                <ErrorBar dataKey="errHigh" direction="y" stroke="#10b981" />
              </Scatter>
              <Tooltip cursor={{ strokeDasharray: "3 3" }} />
            </ScatterChart>
          </ResponsiveContainer>
          <div className="text-xs text-gray-500" data-testid={`reliability-footer-${agentName}`}>
            ECE = {data.ece !== null ? data.ece.toFixed(3) : "—"} • N = {data.n_samples} • Bins = {data.n_bins_used}
          </div>
        </div>
      );
    }
    ```

    ```tsx
    import type { MurphyDecomposition } from "../../api/types";

    interface Props {
      data: MurphyDecomposition | null;
      brier?: number | null;
      agentName: string;
    }

    const TOOLTIP_REL =
      "Reliability (REL) — calibration error. Lower is better. " +
      "Measures how close the predicted probabilities are to the observed frequencies.";
    const TOOLTIP_RES =
      "Resolution (RES) — discrimination. Higher is better. " +
      "Measures how much the agent separates BUYs from SELLs across confidence bins.";
    const TOOLTIP_UNC =
      "Uncertainty (UNC) — climatological base rate. " +
      "Approaches 0.25 when wins occur ~50% of the time (max uncertainty); " +
      "lower when one side dominates.";

    export default function MurphyDecompositionCard({ data, brier, agentName }: Props) {
      if (!data || data.rel === null) {
        return (
          <div
            className="rounded border border-gray-700 bg-gray-900/50 p-3 text-xs text-gray-500 italic"
            data-testid={`murphy-card-empty-${agentName}`}
          >
            Insufficient data for Murphy decomposition (corpus too thin).
          </div>
        );
      }

      const sanityOk =
        brier != null && data.verified_sum != null
          ? Math.abs(data.verified_sum - brier) < 1e-3
          : null;

      return (
        <div
          className="rounded border border-gray-700 bg-gray-900/50 p-3 space-y-2"
          data-testid={`murphy-card-${agentName}`}
        >
          <div className="text-xs font-semibold text-gray-300">Brier decomposition</div>
          <div className="grid grid-cols-3 gap-2 text-xs">
            <div title={TOOLTIP_REL} data-testid={`murphy-rel-${agentName}`}>
              <div className="text-gray-500">REL</div>
              <div className="font-mono text-amber-400">{data.rel?.toFixed(3) ?? "—"}</div>
            </div>
            <div title={TOOLTIP_RES} data-testid={`murphy-res-${agentName}`}>
              <div className="text-gray-500">RES</div>
              <div className="font-mono text-emerald-400">{data.res?.toFixed(3) ?? "—"}</div>
            </div>
            <div title={TOOLTIP_UNC} data-testid={`murphy-unc-${agentName}`}>
              <div className="text-gray-500">UNC</div>
              <div className="font-mono text-gray-300">{data.unc?.toFixed(3) ?? "—"}</div>
            </div>
          </div>
          <div className="text-xs text-gray-500 border-t border-gray-700 pt-1">
            REL − RES + UNC ≈ {data.verified_sum?.toFixed(3) ?? "—"}
            {sanityOk === true && " ✓"}
            {sanityOk === false && " ✗ (Brier mismatch)"}
          </div>
        </div>
      );
    }
    ```

    ```tsx
    import { useState } from "react";
    import type { RestatedDelta } from "../../api/types";

    interface Props {
      deltas: RestatedDelta[] | null | undefined;
      ticker: string;
    }

    export default function RestatedDeltaBadge({ deltas, ticker }: Props) {
      const [showTooltip, setShowTooltip] = useState(false);

      if (!deltas || deltas.length === 0) return null;

      const material = deltas.filter(
        (d) => d.delta_pct !== null && Math.abs(d.delta_pct) > 0.10
      );
      if (material.length === 0) return null;

      const first = material[0];

      return (
        <span className="relative inline-block ml-2" data-testid={`restated-badge-${ticker}`}>
          <button
            type="button"
            onClick={() => setShowTooltip(!showTooltip)}
            onBlur={() => setShowTooltip(false)}
            className="px-1.5 py-0.5 rounded text-xs bg-amber-900/40 text-amber-300 border border-amber-800 hover:bg-amber-900/60"
            aria-label={`Restated values differ from as-filed for ${material.length} metric(s)`}
          >
            Restated
          </button>
          {showTooltip && (
            <div
              role="tooltip"
              className="absolute z-10 mt-1 right-0 w-64 rounded border border-gray-700 bg-gray-900 p-2 text-xs shadow-lg"
              data-testid={`restated-tooltip-${ticker}`}
            >
              <div className="font-semibold mb-1">{first.metric}</div>
              <div className="grid grid-cols-2 gap-2">
                <div>
                  <div className="text-gray-500">As filed</div>
                  <div className="font-mono text-gray-300">
                    {first.as_filed !== null ? first.as_filed.toLocaleString() : "—"}
                  </div>
                </div>
                <div>
                  <div className="text-gray-500">Restated</div>
                  <div className="font-mono text-amber-400">
                    {first.restated !== null ? first.restated.toLocaleString() : "—"}
                  </div>
                </div>
              </div>
              <div className="text-gray-500 mt-1">
                Filed: {first.filing_date ?? "—"}
              </div>
              {material.length > 1 && (

                  +{material.length - 1} other restated metric{material.length > 2 ? "s" : ""}
                </div>
              )}
            </div>
          )}
        </span>
    ```
